ARQUITECTURA GENERAL
====================
Vendedor -> S3 uploads/ -> Lambda validadora -> (ok) / (reject -> rejected/ + SNS)
Cliente -> API Gateway -> Lambda crear orden -> RDS + SQS principal
SQS principal -> Lambda procesadora -> simula pago -> SNS fan-out
SNS fan-out -> SQS pagos
SNS fan-out -> SQS notificación vendedor
SNS fan-out -> Lambda inventario -> RDS
SQS principal DLQ -> EventBridge cada 5 min -> Lambda DLQ -> RDS + SNS admin
Internet -> ALB -> EC2 NGINX -> app principal 70% / canary 30%

---
const AWS = require('aws-sdk');
const s3 = new AWS.S3();
const rekognition = new AWS.Rekognition();
const sns = new AWS.SNS();

const MAX_SIZE = 5 * 1024 * 1024;
const ALLOWED_TYPES = new Set(['image/jpeg', 'image/png']);

exports.handler = async (event) => {
  for (const record of event.Records) {
    const bucket = record.s3.bucket.name;
    const key = decodeURIComponent(record.s3.object.key.replace(/\+/g, ' '));

    if (!key.startsWith('uploads/')) continue;

    const head = await s3.headObject({ Bucket: bucket, Key: key }).promise();

    if (head.ContentLength > MAX_SIZE || !ALLOWED_TYPES.has(head.ContentType)) {
      await rejectFile(bucket, key, 'Invalid size or format');
      continue;
    }

    const moderation = await rekognition.detectModerationLabels({
      Image: { S3Object: { Bucket: bucket, Name: key } },
      MinConfidence: 60
    }).promise();

    if (moderation.ModerationLabels && moderation.ModerationLabels.length > 0) {
      await rejectFile(bucket, key, 'Explicit content detected');
      continue;
    }

    console.log(`Image accepted: ${key}`);
  }

  return { ok: true };
};

async function rejectFile(bucket, key, reason) {
  const fileName = key.split('/').pop();
  const rejectedKey = `rejected/${fileName}`;

  await s3.copyObject({
    Bucket: bucket,
    CopySource: `${bucket}/${encodeURIComponent(key)}`,
    Key: rejectedKey
  }).promise();

  await s3.deleteObject({ Bucket: bucket, Key: key }).promise();

  await sns.publish({
    TopicArn: process.env.SNS_TOPIC_ARN,
    Message: JSON.stringify({
      type: 'image.rejected',
      bucket,
      key: rejectedKey,
      reason
    }),
    MessageAttributes: {
      eventType: { DataType: 'String', StringValue: 'image.rejected' },
      destination: { DataType: 'String', StringValue: 'seller-notifications-queue' }
    }
  }).promise();
}
---

Paso 5: Pegar código
-------------------
1. En Code, reemplaza todo con:

---
const AWS = require('aws-sdk');
const mysql = require('mysql2/promise');
const crypto = require('crypto');

const sqs = new AWS.SQS();

exports.handler = async (event) => {
  const body = typeof event.body === 'string' ? JSON.parse(event.body) : event.body;
  const orderId = crypto.randomUUID();

  const connection = await mysql.createConnection({
    host: process.env.DB_HOST,
    port: Number(process.env.DB_PORT || 3306),
    database: process.env.DB_NAME,
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD
  });

  await connection.execute(
    'INSERT INTO ordenes (order_id, buyer_id, seller_id, producto_id, cantidad, total, estado, payment_status) VALUES (?, ?, ?, ?, ?, ?, "creada", "pendiente")',
    [
      orderId,
      body.buyerId,
      body.sellerId,
      body.productId,
      body.quantity,
      body.amount
    ]
  );
  await connection.end();

  await sqs.sendMessage({
    QueueUrl: process.env.ORDER_QUEUE_URL,
    MessageBody: JSON.stringify({ orderId, ...body }),
    MessageAttributes: {
      eventType: { DataType: 'String', StringValue: 'order.created' },
      destination: { DataType: 'String', StringValue: 'processing' }
    }
  }).promise();

  return {
    statusCode: 201,
    body: JSON.stringify({ message: 'Order created', orderId })
  };
};

const AWS = require('aws-sdk');
const mysql = require('mysql2/promise');

const sns = new AWS.SNS();

exports.handler = async (event) => {
  for (const record of event.Records) {
    const order = JSON.parse(record.body);

    if (order.forceError) {
      throw new Error('Forced failure for DLQ test');
    }

    const paymentStatus = 'paid';

    const connection = await mysql.createConnection({
      host: process.env.DB_HOST,
      port: Number(process.env.DB_PORT || 3306),
      database: process.env.DB_NAME,
      user: process.env.DB_USER,
      password: process.env.DB_PASSWORD
    });

    await connection.execute(
      'UPDATE ordenes SET payment_status = ?, estado = ?, updated_at = NOW() WHERE order_id = ?',
      [paymentStatus, 'pagada', order.orderId]
    );
    await connection.end();

    await publishEvent('payment.completed', 'payment-queue', {
      orderId: order.orderId,
      sellerId: order.sellerId,
      amount: order.amount
    });

    await publishEvent('seller.notification', 'seller-notifications-queue', {
      orderId: order.orderId,
      sellerId: order.sellerId
    });

    await publishEvent('inventory.update', 'inventory-lambda', {
      orderId: order.orderId,
      productId: order.productId,
      quantity: order.quantity
    });
  }

  return { ok: true };
};

async function publishEvent(eventType, destination, detail) {
  await sns.publish({
    TopicArn: process.env.SNS_TOPIC_ARN,
    Message: JSON.stringify({ eventType, destination, detail }),
    MessageAttributes: {
      eventType: { DataType: 'String', StringValue: eventType },
      destination: { DataType: 'String', StringValue: destination }
    }
  }).promise();
}

---
const AWS = require('aws-sdk');
const mysql = require('mysql2/promise');

const sqs = new AWS.SQS();
const sns = new AWS.SNS();

exports.handler = async () => {
  const received = await sqs.receiveMessage({
    QueueUrl: process.env.DLQ_URL,
    MaxNumberOfMessages: 10,
    WaitTimeSeconds: 0,
    MessageAttributeNames: ['All']
  }).promise();

  if (!received.Messages || received.Messages.length === 0) {
    return { processed: 0 };
  }

  for (const message of received.Messages) {
    const payload = message.Body;

    const connection = await mysql.createConnection({
      host: process.env.DB_HOST,
      port: Number(process.env.DB_PORT || 3306),
      database: process.env.DB_NAME,
      user: process.env.DB_USER,
      password: process.env.DB_PASSWORD
    });

    await connection.execute(
      'INSERT INTO errores (source_queue, lambda_name, error_type, message, raw_payload) VALUES (?, ?, ?, ?, ?)',
      [
        'marketaws-order-queue',
        'marketaws-process-order-lambda',
        'DLQ',
        'Message moved to DLQ',
        JSON.stringify({ body: payload })
      ]
    );
    await connection.end();

    await sns.publish({
      TopicArn: process.env.ADMIN_TOPIC_ARN,
      Message: JSON.stringify({
        subject: 'DLQ alert',
        queue: 'marketaws-order-dlq',
        body: payload
      })
    }).promise();

    await sqs.deleteMessage({
      QueueUrl: process.env.DLQ_URL,
      ReceiptHandle: message.ReceiptHandle
    }).promise();
  }

  return { processed: received.Messages.length };
};
